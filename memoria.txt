library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity Mem is
    port(
        clk        : in  std_logic;
        rst        : in  std_logic;
        MemRead    : in  std_logic;
        MemWrite   : in  std_logic;
        End_in     : in  std_logic_vector(31 downto 0); -- Vem do Mux_End
        WriteData  : in  std_logic_vector(31 downto 0); -- Vem do Registrador B
        MemData    : out std_logic_vector(31 downto 0)  -- Vai para IR e MDR
    );
end Mem;

architecture rtl of Mem is
    -- 256 palavras de 32 bits cada
    subtype palavra is std_logic_vector(31 downto 0);
    type ram_type is array (0 to 255) of palavra;
    signal mem : ram_type := (others => (others => '0'));

    -- Índice da palavra: divide o endereço por 4 pegando os bits [9 downto 2]
    signal word_addr : integer range 0 to 255;
begin

    word_addr <= to_integer(unsigned(End_in(9 downto 2)));

    -- Leitura Assíncrona/Combinacional controlada por MemRead
    -- (Permite que no estado S0 o dado saia da memória e já seja gravado no IR na borda do clock)
    MemData <= mem(word_addr) when MemRead = '1' else (others => '0');

    -- Escrita Síncrona na borda de subida do clock
    process(clk, rst)
    begin
        if rst = '1' then
            -- Zera toda a memória em 1 linha
            mem <= (others => (others => '0'));
            
            -- AQUI VOCÊ CARREGA SEU PROGRAMA (Instruções a partir do índice 0)
            -- E SEUS DADOS INICIAIS (Ex: a partir do índice 64 -> endereço 256 em bytes)
            -- mem(0) <= "................................"; -- Instrução 1 (PC = 0)
            -- mem(1) <= "................................"; -- Instrução 2 (PC = 4)
            
        elsif rising_edge(clk) then
            if MemWrite = '1' then
                mem(word_addr) <= WriteData;
            end if;
        end if;
    end process;

end rtl;
